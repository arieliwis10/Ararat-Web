# BEGIN Ararat web estática
AddDefaultCharset UTF-8
ErrorDocument 404 /404.html
DirectoryIndex index.html index.php

<IfModule mod_rewrite.c>
  RewriteEngine On
  # Forzar HTTPS
  RewriteCond %{HTTPS} !=on
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
  # www -> dominio sin www (una sola URL canónica para SEO)
  RewriteCond %{HTTP_HOST} ^www\.araratchile\.com$ [NC]
  RewriteRule ^ https://araratchile.com%{REQUEST_URI} [L,R=301]
</IfModule>

<IfModule mod_headers.c>
  # Archivos con hash en el nombre (CSS/JS/imágenes de Astro): caché de 1 año
  <If "%{REQUEST_URI} =~ m#^/_astro/#">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </If>
  # Video y favicon: 1 semana
  <FilesMatch "\.(mp4|svg|ico)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  # HTML: siempre revalidar, para que un deploy se vea de inmediato
  <FilesMatch "\.html$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml application/xml text/plain
</IfModule>

<IfModule mod_mime.c>
  AddType image/avif .avif
  AddType image/webp .webp
  AddType video/mp4 .mp4
</IfModule>
# END Ararat web estática
